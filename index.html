flick-api.py 
from fastapi import FastAPI
from services import tmdb_engine

app = FastAPI(title="Flick Core Engine")

@app.get("/api/movies/swipe")
async def get_swipe_deck(user_id: str):
    recommendations = await tmdb_engine.fetch(user_id)
    return {"status": "success", "data": recommendations}
